{{/* External links (external domains, including protocol-relative //) open in a new tab */}} {{- $dest := .Destination -}} {{- $siteHost := .Page.Site.BaseURL | strings.TrimPrefix "http://" | strings.TrimPrefix "https://" | strings.TrimSuffix "/" -}} {{- $destNorm := $dest | strings.TrimPrefix "http://" | strings.TrimPrefix "https://" | strings.TrimPrefix "//" -}} {{- $destHost := index (split "/" $destNorm) 0 -}} {{- $isExternal := and (or (hasPrefix $dest "http") (hasPrefix $dest "//")) (ne $destHost $siteHost) -}} {{ .Text | safeHTML }}
